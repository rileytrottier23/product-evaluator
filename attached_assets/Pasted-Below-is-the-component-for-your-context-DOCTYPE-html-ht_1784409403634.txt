  // Stack Card Intersection Logic
  const initStackCards = () => {
    const stackContainer = document.getElementById('cta-stack-container');
    if (stackContainer) {
      const stackObserver = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            const layers = stackContainer.querySelectorAll('[data-stack-layer]');
            layers.forEach((layer) => {
              const depth = parseInt(layer.dataset.stackLayer);
              layer.classList.remove('opacity-0');
              if(depth === 0) {
                layer.style.transform = 'translateY(0)';
              } else {
                const x = depth * -18;
                const y = depth * 15;
                layer.style.transform = `translate(${x}px, ${y}px)`;
              }
            });
            stackObserver.unobserve(stackContainer);
          }
        });
      }, { threshold: 0.2 });
      stackObserver.observe(stackContainer);
    }
  };
  initStackCards();
});

    document.addEventListener('DOMContentLoaded', () => {

      const mobileMenu = document.getElementById('mobile-menu');
      const toggle = document.getElementById('mobile-menu-toggle');
      const close = document.getElementById('mobile-menu-close');
      const links = document.querySelectorAll('.mobile-nav-link');

      const openMenu = () => {
        mobileMenu.classList.remove('translate-x-full');
        document.body.style.overflow = 'hidden';
      };

      const closeMenu = () => {
        mobileMenu.classList.add('translate-x-full');
        document.body.style.overflow = '';
      };

      toggle?.addEventListener('click', openMenu);
      close?.addEventListener('click', closeMenu);
      links.forEach(link => link.addEventListener('click', closeMenu));

      const observer = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            if (entry.target.hasAttribute('data-animation-top-down')) {
              entry.target.classList.add('animate-top-down-visible');
            } else if (entry.target.hasAttribute('data-animation-text-reveal')) {
              entry.target.classList.add('animate-text-reveal-visible');
            } else {
              entry.target.classList.add('animate-on-scroll-visible');
            }
            observer.unobserve(entry.target);
          }
        });
      }, { threshold: 0.1 });

      document.querySelectorAll('[data-animation-on-scroll]').forEach(el => {
        el.classList.add('animate-on-scroll-hidden');
        observer.observe(el);
      });

      document.querySelectorAll('[data-animation-top-down]').forEach(el => {
        el.classList.add('animate-top-down-hidden');
        observer.observe(el);
      });

      document.querySelectorAll('[data-animation-text-reveal]').forEach(el => {
        el.classList.add('animate-text-reveal-hidden');
        observer.observe(el);
      });
    });
